const ROWS = 12;

const COLS = 6;

let turn = 0;

let blueScore = 0;

let redScore = 0;

let gameSeconds = 180;

let playerSeconds = 15;


/* DOM */

const boardElement =
    document.getElementById("board");

const explodeSound =
    document.getElementById("explodeSound");

const turnText =
    document.getElementById("turnText");


/* BOARD DATA */

const board =
    Array.from(
        { length: ROWS },

        () =>
            Array.from(
                { length: COLS },

                () => ({
                    color: 0,
                    count: 0
                })
            )
    );


/* CREATE BOARD */

function createBoard()
{
    for (let row = 0; row < ROWS; row++)
    {
        for (let col = 0; col < COLS; col++)
        {
            const cell =
                document.createElement("div");

            cell.classList.add("cell");

            cell.dataset.row = row;
            cell.dataset.color=-1;

            cell.dataset.col = col;

            cell.addEventListener(
                "click",
                handleMove
            );

            boardElement.appendChild(cell);
        }
    }
}

createBoard();


/* TIMER */

setInterval(() =>
{
    gameSeconds--;

    playerSeconds--;

    const gm =
        Math.floor(gameSeconds / 60);

    const gs =
        String(gameSeconds % 60)
            .padStart(2, "0");

    const ps =
        String(playerSeconds)
            .padStart(2, "0");

    document.getElementById(
        "gameTimer"
    ).innerText =
        `GAME TIMER ${gm}:${gs}`;

    document.getElementById(
        "playerTimer"
    ).innerText =
        `PLAYER TIMER 0:${ps}`;

    if (playerSeconds <= 0)
    {
        switchTurn();
    }

}, 1000);


function cornerCell(row,col){
    return row===0 && col===0 || row===0 && col ===COLS-1 
        || row === ROWS-1 && col===0 || row ===ROWS-1 && col===COLS-1;
}
function edgeCell(row,col){
    return row===0 && (col >0 ||col <COLS-1) || row===ROWS-1 && (col>0 || col<COLS-1)
        ||  col===0 && ( row >0 || row < ROWS-1)  || col===0 && ( row >0 || row <ROWS-1);
}
function innerCell(row,col){
    return !cornerCell(row,col) && !edgeCell(row,col);
}




/* HANDLE MOVE */

function initialiseBalls(row,col,cell){
    if(cornerCell(row,col)){
        cell.dataset.count=1;
    }
    else if(edgeCell(row,col)){
        cell.dataset.count=2;
    }
    else{
        cell.dataset.count=3;
    }
}

function firstMove(row,col){
    const cell =
    board[row][col];
    cell.dataset.color=turn%2;
    if(turn===0){
        initialiseBalls(row,col,cell);
    }else{
        initialiseBalls(row,col,cell);
    }
}

function handleMove(event)
{
    const row =
        parseInt(
            event.target.dataset.row
        );

    const col =
        parseInt(
            event.target.dataset.col
        );

    const cell =
        board[row][col];
    
    if(turn===0||turn===1){
        firstMove(row,col);
        cell.color = turn;
        updateBoard();
        switchTurn();
        return;
    }    

    if (
        cell.color ===-1 &&
        cell.color !== (turn%2)
    )
    {
        return;
    }

    cell.color = turn%2;

    cell.count++;

    updateBoard();

    chainReaction(row, col);

    switchTurn();
}


/* CAPACITY */

function getCapacity(row, col)
{

    if (cornerCell(row,col))
    {
        return 1;
    }
    else if (edgeCell(row,col))
    {
        return 2;
    }
    return 3;
}


/* CHAIN REACTION */

function chainReaction(row, col)
{
    const cell =
        board[row][col];

    if (
        cell.count <=
        getCapacity(row, col)
    )
    {
        return;
    }

    explodeSound.currentTime = 0;

    explodeSound.play();

    const color =
        cell.color;

    cell.color = 0;

    cell.count = 0;

    const directions =
    [
        [-1, 0],
        [1, 0],
        [0, -1],
        [0, 1]
    ];

    for (const [dr, dc] of directions)
    {
        const nr = row + dr;

        const nc = col + dc;

        if (
            nr >= 0 &&
            nr < ROWS &&
            nc >= 0 &&
            nc < COLS
        )
        {
            board[nr][nc].color =
                color;

            board[nr][nc].count++;

            chainReaction(nr, nc);
        }
    }

    updateBoard();
}


/* UPDATE BOARD */

function updateBoard()
{
    const cells =
        document.querySelectorAll(".cell");

    blueScore = 0;

    redScore = 0;

    cells.forEach(cell =>
    {
        const row =
            parseInt(cell.dataset.row);

        const col =
            parseInt(cell.dataset.col);

        const data =
            board[row][col];

        cell.innerHTML = "";

        if (data.color === 1)
        {
            blueScore += data.count;
        }

        else if (data.color === 2)
        {
            redScore += data.count;
        }

        const colorClass =
            data.color === 1
                ? "blue-orb"
                : "red-orb";

        /* 1 ORB */

        if (data.count === 1)
        {
            const orb =
                createOrb(colorClass);

            orb.style.left = "50%";

            orb.style.top = "50%";

            orb.style.transform =
                "translate(-50%, -50%)";

            cell.appendChild(orb);
        }

        /* 2 ORBS */

        else if (data.count === 2)
        {
            const orb1 =
                createOrb(colorClass);

            const orb2 =
                createOrb(colorClass);

            orb1.style.left = "35%";
            orb1.style.top = "50%";

            orb2.style.left = "65%";
            orb2.style.top = "50%";

            orb1.style.transform =
                "translate(-50%, -50%)";

            orb2.style.transform =
                "translate(-50%, -50%)";

            cell.appendChild(orb1);

            cell.appendChild(orb2);
        }

        /* 3 ORBS */

        else if (data.count >= 3)
        {
            const positions =
            [
                ["50%", "28%"],
                ["32%", "68%"],
                ["68%", "68%"]
            ];

            positions.forEach(pos =>
            {
                const orb =
                    createOrb(colorClass);

                orb.style.left =
                    pos[0];

                orb.style.top =
                    pos[1];

                orb.style.transform =
                    "translate(-50%, -50%)";

                cell.appendChild(orb);
            });
        }
    });

    document.getElementById(
        "blueScore"
    ).innerText =
        `BLUE ${blueScore}`;

    document.getElementById(
        "redScore"
    ).innerText =
        `RED ${redScore}`;
}


/* CREATE ORB */

function createOrb(colorClass)
{
    const orb =
        document.createElement("div");

    orb.classList.add(
        "orb",
        colorClass
    );

    return orb;
}


/* SWITCH TURN */

function switchTurn()
{
    turn++;

    playerSeconds = 15;

    if (turn%2 === 0)
    {
        turnText.innerText =
            "BLUE player's turn";
        turnText.style.color="blue";
    }
    
    else
        {
            turnText.innerText =
            "RED player's turn";
            turnText.style.color="red";
    }
}
